// ============================================================ // QuickMemo GAS v5 - スプレッドシート連動 + Googleカレンダー連動 // ============================================================ const SHEET_NAME = 'QuickMemo'; const NOTIFY_MINUTES_BEFORE = 15; function doGet(e) { return HtmlService.createHtmlOutputFromFile('index') .setTitle('QuickMemo') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } function doPost(e) { try { const data = JSON.parse(e.postData.contents); // カレンダー登録リクエスト if (data.action === 'addCalendar') { addToCalendar(data); return ContentService.createTextOutput(JSON.stringify({ status: 'ok', action: 'calendar' })) .setMimeType(ContentService.MimeType.JSON); } // 通常のメモ・ToDoはスプレッドシートに保存 writeToSheet(data); return ContentService.createTextOutput(JSON.stringify({ status: 'ok' })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.message })) .setMimeType(ContentService.MimeType.JSON); } } // スプレッドシートに書き込む function writeToSheet(item) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(['ID', 'タイプ', 'タイトル', '内容', '期限/開始日', '終了日', '完了', '作成日時', '最終更新', '通知済み']); sheet.getRange(1, 1, 1, 10).setFontWeight('bold'); sheet.setFrozenRows(1); } const data = sheet.getDataRange().getValues(); const ids = data.slice(1).map(r => r[0]); const rowIndex = ids.indexOf(item.id); const typeLabel = item.type === 'todo' ? 'ToDo' : item.type === 'cal' ? 'カレンダー' : 'メモ'; const dateField = item.type === 'cal' ? item.calStart : item.deadline; const endField = item.type === 'cal' ? item.calEnd : ''; const rowData = [ item.id, typeLabel, item.title, item.type === 'cal' ? (item.calDesc || '') : (item.body || ''), dateField ? new Date(dateField) : '', endField ? new Date(endField) : '', item.done ? '完了' : '', item.createdAt ? new Date(item.createdAt) : new Date(), new Date(), '' ]; if (rowIndex >= 0) { sheet.getRange(rowIndex + 2, 1, 1, 9).setValues([rowData.slice(0, 9)]); } else { sheet.appendRow(rowData); } const lastRow = sheet.getLastRow(); if (lastRow >= 2) { sheet.getRange(2, 5, lastRow - 1, 2).setNumberFormat('yyyy/mm/dd hh:mm'); sheet.getRange(2, 8, lastRow - 1, 2).setNumberFormat('yyyy/mm/dd hh:mm'); sheet.autoResizeColumns(1, 10); } } // Googleカレンダーに追加する function addToCalendar(item) { const calendar = CalendarApp.getDefaultCalendar(); const title = item.title || '無題'; const desc = item.calDesc || ''; let startDate, endDate; if (item.calStart) { startDate = new Date(item.calStart); } else { startDate = new Date(); } if (item.calEnd) { endDate = new Date(item.calEnd); } else { endDate = startDate; } // 開始・終了が同じ場合は1時間後を終了に if (startDate.getTime() === endDate.getTime()) { endDate = new Date(startDate.getTime() + 60 * 60 * 1000); } calendar.createEvent(title, startDate, endDate, { description: desc }); } // リマインダー（毎分実行） function checkDeadlines() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) return; const data = sheet.getDataRange().getValues(); const now = new Date(); const notifyBefore = NOTIFY_MINUTES_BEFORE * 60 * 1000; const userEmail = Session.getActiveUser().getEmail(); data.slice(1).forEach((row, i) => { const type = row[1]; const title = row[2]; const body = row[3]; const deadline = row[4]; const done = row[6]; const notified = row[9]; if (type !== 'ToDo' || done === '完了' || !deadline || notified) return; const diff = new Date(deadline) - now; if (diff > 0 && diff <= notifyBefore) { const subject = '[QuickMemo] リマインダー: ' + title; const message = '期限が近づいています。\n\nタイトル: ' + title + '\n内容: ' + (body || '（なし）') + '\n期限: ' + Utilities.formatDate(new Date(deadline), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm') + '\n\nこのメールはQuickMemoから自動送信されています。'; GmailApp.sendEmail(userEmail, subject, message); sheet.getRange(i + 2, 10).setValue('通知済み'); } }); } // ============================================================ // セットアップ手順（v5追加分） // ============================================================ // カレンダー連携を使う場合、スクリプトにカレンダーの権限を追加する必要があります。 // 1. GASエディタ左メニュー「サービス」の「+」 // 2. 「Google Calendar API」を追加 // 3. 再デプロイ（デプロイを管理→新しいバージョン） // 4. 承認画面でカレンダーへのアクセスを許可 // ============================================================
